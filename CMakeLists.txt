cmake_minimum_required(VERSION 3.20)
project(StockGuard VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(STOCKGUARD_ASAN "Enable address and undefined behavior sanitizers" OFF)
option(STOCKGUARD_TSAN "Enable thread sanitizer" OFF)
if(STOCKGUARD_ASAN AND STOCKGUARD_TSAN)
  message(FATAL_ERROR "Select one sanitizer mode")
endif()
if(STOCKGUARD_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(STOCKGUARD_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
include(FetchContent)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03 URL_HASH SHA256=0dbc5e40a01ff142e7e68c03e85247a4dcede2f592d12d3677dee3664d17975a DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
option(STOCKGUARD_SYSTEM_SQLITE "Use a patched system SQLite >= 3.51.3" OFF)
if(STOCKGUARD_SYSTEM_SQLITE)
  find_package(SQLite3 3.51.3 REQUIRED)
else()
  FetchContent_Declare(sqlite URL https://sqlite.org/2026/sqlite-amalgamation-3540000.zip
    URL_HASH SHA256=68e913b0fe8ec6b4e5f391c594a28872884e80d6cd042a1e50c957be46b5aa4d
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(sqlite)
  add_library(stockguard_sqlite STATIC ${sqlite_SOURCE_DIR}/sqlite3.c)
  target_include_directories(stockguard_sqlite PUBLIC ${sqlite_SOURCE_DIR})
  target_compile_definitions(stockguard_sqlite PRIVATE SQLITE_THREADSAFE=1)
  target_link_libraries(stockguard_sqlite PUBLIC ${CMAKE_DL_LIBS})
  add_library(SQLite::SQLite3 ALIAS stockguard_sqlite)
endif()
find_package(Threads REQUIRED)
add_library(stockguard_core src/domain.cpp src/database.cpp src/store.cpp)
target_include_directories(stockguard_core PUBLIC include)
target_link_libraries(stockguard_core PUBLIC nlohmann_json::nlohmann_json SQLite::SQLite3 Threads::Threads)
target_compile_options(stockguard_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(stockguard src/main.cpp)
target_link_libraries(stockguard PRIVATE stockguard_core)
target_compile_options(stockguard PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/6910c9d9165801d8827d628cb72eb7ea9dd538c5 URL_HASH SHA256=bde221be7f3841fcbc3971665d77d717116394a42155d988ee6407dfc39f1f09 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googletest)
  add_executable(stockguard_tests tests/engine_test.cpp tests/store_test.cpp)
  target_link_libraries(stockguard_tests PRIVATE stockguard_core GTest::gtest_main)
  target_compile_options(stockguard_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  include(GoogleTest)
  if(STOCKGUARD_TSAN)
    gtest_discover_tests(stockguard_tests PROPERTIES ENVIRONMENT
      "TSAN_OPTIONS=suppressions=${CMAKE_CURRENT_SOURCE_DIR}/config/tsan.supp")
  else()
    gtest_discover_tests(stockguard_tests)
  endif()
endif()
add_executable(stockguard_benchmark benchmarks/benchmark.cpp)
target_link_libraries(stockguard_benchmark PRIVATE stockguard_core)
target_compile_options(stockguard_benchmark PRIVATE -Wall -Wextra -Wpedantic -Werror)
